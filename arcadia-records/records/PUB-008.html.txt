<p style="text-align: center;"><span style="font-size: 18px;"><strong>Update Regarding Proposed Protected Tree Removal at Santa Anita Park</strong></span></p>
<div style="text-align: justify;">At the request of Santa Anita Park, the Applicant, <strong>processing of protected tree removal permit TRH 26-11/TRE 26-09 has been placed on hold</strong>. Santa Anita Park has shared that work is already underway with the International Equestrian Federation (FEI) course design team, LA28, and construction engineers to further evaluate opportunities to address potential impacts to protected trees while preparing for the 2028 Olympic and Paralympic Games.</div>
<div style="text-align: justify;">&nbsp;</div>
<div style="text-align: justify;">Since review of the permit application has been paused, the City&rsquo;s administrative review process is also on hold. At this time, there is<strong> </strong>no decision regarding the proposed tree removals pending before City staff or the Arcadia City Council. The City will continue to follow all review and public notification processes should a revised proposal be submitted. If a revised permit application is submitted, the public notification process will begin again to allow the community to review and provide input on the revised plans. Updates and relevant information will be made available on the project webpage, where interested parties can subscribe to project updates:&nbsp; <a rel="noopener" href="https://www.ArcadiaCA.gov/oaktreesatccc" target="_blank">www.ArcadiaCA.gov/oaktreesatccc</a>.</div>

<script>
var translateURL = "//translate.google.com/translate_a/element.js?cb=googleTranslateElementInit";
$.getScript(translateURL);

function clickLanguage(langText, maxTries, intervalMs) {
	langText = (langText || '').trim();
	maxTries = maxTries || 30;
	intervalMs = intervalMs || 200;
	var tries = 0;
	var timer = setInterval(function () {
		var $frame = $("iframe.skiptranslate");
		if ($frame.length) {
			var doc = $frame.contents();
			var $target = doc.find('span.text').filter(function () {
				return $(this).text().trim() === langText;
			}).closest('a');
			if (!$target.length) {
				$target = doc.find('span.text').filter(function () {
					return $(this).text().trim() === 'English';
				}).closest('a');
			}
			if ($target.length) {
				$target.get(0).click();
				clearInterval(timer);
				return;
			}
		}
		tries++;
		if (tries >= maxTries) clearInterval(timer);
	}, intervalMs);
}

function getCookie(cname) {
	let name = cname + "=";
	let ca = document.cookie.split(';');
	for(let i = 0; i < ca.length; i++) {
		let c = ca[i];
		while (c.charAt(0) == ' ') c = c.substring(1);
		if (c.indexOf(name) == 0) return c.substring(name.length, c.length);
	}
	return "";
}

function sanitizeFlag(code) {
	var v = (code || "").toLowerCase().trim();
	if (!v || v === "undefined" || v === "null") return "";
	if (!/^[a-z]{2}$/.test(v)) return "";
	return v;
}

function applyActiveFromCookie() {
	var code = sanitizeFlag(getCookie('lang')) || sanitizeFlag(getCookie('code')) || 'us';
	var $btn = $('.translation-links button').filter(function(){
		var dataCode = ($(this).data('code') || '').toString().toLowerCase();
		if (dataCode && dataCode === code) return true;
		var cls = ($(this).find('span[class*="fi-"]').attr('class') || '');
		var m = cls.match(/\bfi-([a-z]{2})\b/i);
		return m && m[1].toLowerCase() === code;
	}).first();
	if (!$btn.length && code === 'us') $btn = $('.translation-links button[data-lang="English"]').first();
	$('.translation-links button').removeClass('active');
	if ($btn.length) $btn.addClass('active');
}

(function initSelectedFlag(){
	var initialCode = sanitizeFlag(getCookie('lang')) || sanitizeFlag(getCookie('code')) || "us";
	$('.selected-flag').attr('class', 'fi fi-' + initialCode + ' selected-flag');
	$('.c-code').text(initialCode);
	applyActiveFromCookie();
})();

$(".translation-links button").on("click", function () {
	var lang = $(this).data("lang");
	$('.translation-links button').removeClass('active');
	$(this).addClass('active');
	clickLanguage(lang);
	var cCode = $(this).data('code');
	if (!cCode) {
		var cls = $(this).find('span[class*="fi-"]').attr('class') || '';
		var m = cls.match(/\bfi-([a-z]{2})\b/i);
		if (m) cCode = m[1].toLowerCase();
	}
	if (!cCode) cCode = 'us';
	$('.selected-flag').attr('class', 'fi fi-' + cCode + ' selected-flag');
	$('.c-code').text(cCode);
	document.cookie = 'lang=' + cCode + '; path=/';
	document.cookie = 'code=' + cCode + '; path=/';
	return false;
});

var $translateReset = $(".translate-reset");
$translateReset.on("keydown click", function (e) {
	if (e.keyCode === 13 || e.type === "click") {
		clickLanguage('English');
		$('.translation-links button').removeClass('active');
		$('.translation-links button[data-lang="English"]').addClass('active');
		$('.selected-flag').attr('class','fi fi-us selected-flag');
		$('.c-code').text('us');
		document.cookie = 'lang=us; path=/';
		document.cookie = 'code=us; path=/';
	}
});

(function($){
	var $form = $('.translate-search');
	var $input = $form.find('input[type="search"]');
	var $list = $('.translation-links');
	var $btns = $list.find('div');
	var $empty = $('.lang-no-results');
	var $status = $('.lang-search-status');
	$form.on('submit', function(e){ e.preventDefault(); });
	function norm(s){
		return (s||'').toString().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'');
	}
	$btns.each(function(){
		var $b = $(this);
		var txt = $b.text().trim();
		var lang = ($b.data('lang')||'')+'';
		$b.attr('data-search', norm(txt + ' ' + lang));
	});
	function updateStatus(visible, total){
		if ($status.length) $status.text(visible + ' of ' + total + ' languages');
		if ($empty.length)  $empty.toggleClass('d-none', visible > 0);
	}
	var total = $btns.length;
	updateStatus(total, total);
	function filter(){
		var q = norm($input.val());
		if (!q){
			$btns.removeClass('d-none');
			updateStatus(total, total);
			return;
		}
		var visible = 0;
		$btns.each(function(){
			var $b = $(this);
			var hay = $b.attr('data-search') || '';
			var match = hay.indexOf(q) !== -1;
			$b.toggleClass('d-none', !match);
			if (match) visible++;
		});
		$btns.filter('.d-none').attr('tabindex', -1);
		$btns.not('.d-none');
		updateStatus(visible, total);
	}
	$input.on('input keyup', filter);
})(jQuery);

			} else {
				allLabelsSet = false;
			}
		});

		if (allLabelsSet) {
			googleObserver.disconnect();
		}
	}

		for (let mutation of mutationsList) {

var catalogueDetailsMap = { modal: null };
var catalogueJSONData = null;

if (typeof indexByIdOrRid !== 'function') {
	function indexByIdOrRid(data){
		if (!data) return {};
		if (!Array.isArray(data)) return data;
		return data.reduce((m, item) => {
			if (!item) return m;
			const key = item.rid ?? item.id;
			if (key != null) m[String(key)] = item;
			return m;
		}, {});
	}
}

function openCatalogueDetails(rid) {
	if (!catalogueDetailsMap.modal || !catalogueJSONData) return;

	rid = String(rid);
	var cRecord = catalogueJSONData[rid];
	if (!cRecord) return;

	var m = catalogueDetailsMap;

	var parsedOptions = cRecord.options;
	if (typeof parsedOptions === 'string') {
		try { parsedOptions = JSON.parse(parsedOptions); } catch(e){ parsedOptions = {}; }
	}
	parsedOptions = parsedOptions || {};

	var imgURL = (cRecord.image || '').trim();
	if (imgURL && !imgURL.endsWith('/')) {
		m.img && m.img.classList.remove('d-none');
		if (m.img) {
			m.img.style.backgroundImage = "url('" + imgURL + "')";
		}
	} else {
		if (m.img) {
			m.img.style.backgroundImage = '';
			m.img.classList.add('d-none');
		}
	}

	// Title
	if (m.title) {
		var rawTitle = cRecord.title || cRecord.name || "";
		m.title.textContent = decodeHtmlEntities(rawTitle);
	}

	if (m.helpful) {
		var links = m.helpful.querySelectorAll("a");
		if (links.length) {
			var el = links[0];
			var webUrl  = (parsedOptions.website_url || "").trim();
			var webText = (parsedOptions.website_text || "Visit Website").trim();
			if (webUrl && webUrl !== "#") {
				el.href = webUrl;
				el.target = "_blank";
				var span = el.querySelector("span");
				if (span) span.innerText = webText;
				el.parentElement && el.parentElement.classList.remove("d-none");
			} else {
				el.href = "";
				el.parentElement && el.parentElement.classList.add("d-none");
			}
		}
	}

	if (m.maplink) {
		var mapAs = m.maplink.querySelectorAll("a");
		if (mapAs.length) {
			var mapEl = mapAs[0];
			var addr = (parsedOptions.val_address || "").trim();
			if (addr) {
				var encoded = encodeURIComponent(addr).replace(/%20/g, "+");
				mapEl.href = "https://www.google.com/maps/search/?api=1&query=" + encoded;
				mapEl.target = "_blank";
				mapEl.parentElement && mapEl.parentElement.classList.remove('d-none');
			} else {
				mapEl.href = "";
				mapEl.parentElement && mapEl.parentElement.classList.add('d-none');
			}
		}
	}

	if (m.contacts) {
		m.contacts.innerHTML = "";
		if (Array.isArray(parsedOptions.contacts)) {
			parsedOptions.contacts.forEach(function(contact) {
				var newAddress = document.createElement("address");
				newAddress.innerHTML = '<strong class="small text-muted">' + (contact.val_header||'') + '</strong><br>';

				var inner = contact.val_text || '';
				if (contact.val_href) {
					try {
						var href = contact.val_href;
						var isSelf = href.startsWith('tel:') || href.startsWith('mailto:') || href.startsWith(location.origin);
						inner = '<a href="'+href+'" target="'+(isSelf?'_self':'_blank')+'">'+ inner +'</a>';
					} catch(e) { /* ignore bad href */ }
				}
				newAddress.innerHTML += inner;
				m.contacts.append(newAddress);
			});
		}
	}

	if (m.hours) {
		const table = m.hours; 
		const dayKeys = [
			'val_sunday','val_monday','val_tuesday','val_wednesday',
			'val_thursday','val_friday','val_saturday'
		];
	
		const tds = Array.from(table.querySelectorAll('tbody td'));
		let hasAny = false;
	
		tds.forEach((td, i) => {
			const val = parsedOptions.hours && parsedOptions.hours[dayKeys[i]];
			const v = (val || '').trim();
			td.textContent = v;
			if (v) hasAny = true;
		});
	
		table.classList.toggle('d-none', !hasAny);
	}

	if (m.detail) {
		const pane = m.detail;
		const tabPane = pane.closest('.tab-pane');
		const tabLink = tabPane ? document.querySelector(`[data-bs-target="#${tabPane.id}"]`) : null;
	
		pane.innerHTML = "Loading...";
	
		fetch('catalogue/' + encodeURIComponent(rid) + '.php?display=json')
			.then(res => res.json())
			.then(data => {
				const html = (data && data.details) ? data.details.trim() : "";
	
				pane.innerHTML = html || "";
	
				const isEmpty = !html;
				if (tabPane) tabPane.classList.toggle('d-none', isEmpty);
				if (tabLink) tabLink.classList.toggle('d-none', isEmpty);
			})
			.catch(() => {
				pane.innerHTML = "";
				if (tabPane) tabPane.classList.add('d-none');
				if (tabLink) tabLink.classList.add('d-none');
			});
	}

	// Gallery #TODO)
	if (m.gallery) m.gallery.innerHTML = "";

	try {
		bootstrap.Offcanvas.getOrCreateInstance(m.modal).show();
	} catch(e) {
		console.error('Offcanvas show failed:', e);
	}

	if (window.pushOcState) pushOcState('catalogueDetails', rid);
}

function hasImg(url) {
	if (!url) return false;
	var s = String(url).trim();
	if (!s) return false;
	if (/^(null|undefined|none|n\/a)$/i.test(s)) return false;
	return true;
}

/* global helper: decode &#10024; &amp; etc */
window.decodeHtmlEntities = window.decodeHtmlEntities || function (str) {
	if (str == null) return '';
	const ta = document.createElement('textarea');
	ta.innerHTML = String(str);
	return ta.value;
};

function setFeaturedImage(url) {
	var m = newsDetailsMap;
	var modal = m.modal;
	if (!modal) return;

	var wrap = modal.querySelector(".featured-image-wrapper");
	if (!wrap) return;

	if (hasImg(url)) {
		wrap.className = "featured-image-wrapper ratio ratio-16x9 mb-4 featured-image-wrap rounded";
		wrap.style.backgroundImage = "url('" + url + "')";
		m.img = wrap;
	} else {
		wrap.className = "featured-image-wrapper";
		wrap.style.backgroundImage = "";
		m.img = null;
	}
}

<script>
(function () {
  var sectionSelectors = ['.rz-page-index #hero-section', '.rz-page-index #news-section', '.rz-page-index #calendar-section'];
  var sections = sectionSelectors
    .map(function (sel) { return document.querySelector(sel); })
    .filter(function (el) { return !!el; });

  if (sections.length < 2) return;

  var current    = 0;
  var animating  = false;
  var deltaStore = 0;
  var threshold  = 80;
  var animMs     = 800;

  function getScrollY() {
    return window.pageYOffset || document.documentElement.scrollTop || 0;
  }

  function headerOffset() {
    var h = document.querySelector('.site-header-wrap') ||
            document.querySelector('.site-header');
    return h ? h.offsetHeight : 0;
  }

  function findCurrentIndex() {
    var mid = getScrollY() + window.innerHeight / 2;
    var idx = 0;
    var offset = headerOffset();

    for (var i = 0; i < sections.length; i++) {
      var rect = sections[i].getBoundingClientRect();
      var top  = rect.top + getScrollY() - offset;
      if (mid >= top) idx = i;
    }
    current = idx;
  }

  function setHashFromIndex(idx) {
    if (!sections[idx] || !sections[idx].id) return;
    var hash = '#' + sections[idx].id;
    if (history && history.replaceState) {
      history.replaceState(null, '', location.pathname + location.search + hash);
    } else {
      location.hash = hash;
    }
  }

  function smoothScrollTo(yTarget, cb) {
    var startY    = getScrollY();
    var distance  = yTarget - startY;
    var startTime = null;

    function step(timestamp) {
      if (!startTime) startTime = timestamp;
      var progress = Math.min((timestamp - startTime) / animMs, 1);
      var eased = 0.5 - Math.cos(progress * Math.PI) / 2;
      var newY  = startY + distance * eased;

      window.scrollTo(0, Math.round(newY));

      if (progress < 1) {
        window.requestAnimationFrame(step);
      } else {
        if (typeof cb === 'function') cb();
      }
    }

    window.requestAnimationFrame(step);
  }

  function snapTo(index) {
    if (index < 0 || index >= sections.length) return;

    var targetEl = sections[index];
    if (!targetEl) return;

    animating  = true;
    deltaStore = 0;

    var offset = headerOffset();
    var rect   = targetEl.getBoundingClientRect();
    var targetY = Math.max(0, Math.round(rect.top + getScrollY() - offset));

    document.documentElement.style.scrollBehavior = 'auto';
    document.body.style.scrollBehavior = 'auto';

    smoothScrollTo(targetY, function () {
      current   = index;
      animating = false;
      setHashFromIndex(current);
    });
  }

  function isInOpenOverlay(target) {
    if (!target || !target.closest) return false;
    return !!target.closest('.offcanvas.show, .modal.show');
  }

  function onWheel(event) {
    // Desktop only snap
    if (window.innerWidth < 992) return;

    var e = event || window.event;

    if (isInOpenOverlay(e.target)) return;

    var oe = e.originalEvent || e;
    var deltaY = 0;

    if (typeof oe.deltaY !== 'undefined') {
      deltaY = oe.deltaY;
    } else if (typeof oe.wheelDelta !== 'undefined') {
      deltaY = -oe.wheelDelta;
    } else if (typeof oe.detail !== 'undefined') {
      deltaY = oe.detail;
    }

    if (!deltaY) return;

    if (animating) {
      if (e.preventDefault) e.preventDefault();
      e.returnValue = false;
      return;
    }

    if (e.preventDefault) e.preventDefault();
    e.returnValue = false;

    deltaStore += deltaY;

    if (Math.abs(deltaStore) >= threshold) {
      var dir  = deltaStore > 0 ? 1 : -1;
      var next = Math.max(0, Math.min(sections.length - 1, current + dir));
      deltaStore = 0;

      if (next !== current) snapTo(next);
    }
  }

  findCurrentIndex();

  window.addEventListener('wheel',          onWheel, { passive: false });
  window.addEventListener('mousewheel',     onWheel, { passive: false });
  window.addEventListener('DOMMouseScroll', onWheel, { passive: false });

  window.addEventListener('resize', function () {
    if (!animating) findCurrentIndex();
  });

  window.addEventListener('load', function () {
    if (window.innerWidth < 992) return;
    var hash = location.hash;
    if (!hash) return;

    for (var i = 0; i < sections.length; i++) {
      if ('#' + sections[i].id === hash) {
        snapTo(i);
        break;
      }
    }
  });
})();
</script>
